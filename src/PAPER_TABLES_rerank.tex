% ===================== Algorithm ==========================================
\begin{algorithm}[t]
\caption{Non-shortening, length-normalised pool rescoring}
\label{alg:rescoring}
\begin{algorithmic}[1]
\Require pool $\mathcal{C}$ with forced-CTC scores; front-end hypothesis $h$;
         prompt hotwords $\mathcal{H}$
\Ensure  $h'$ with $|h'| \ge |h|$ and $P \subseteq h'$
\State $P \leftarrow \{\, p \in \mathcal{H} : p \subseteq h \,\}$
\State $\mathcal{S} \leftarrow \{\, c \in \mathcal{C} : \forall p \in P,\; p \subseteq c \,\}$
\State $\mathcal{S} \leftarrow \{\, c \in \mathcal{S} : |c| \ge |h| \,\}$
         \Comment{$h \in \mathcal{S}$, so $\mathcal{S} \neq \emptyset$}
\State $h' \leftarrow \arg\max_{c \in \mathcal{S}} \bigl( \mathrm{ews}(c),\; \mathrm{ctc}(c) / n_{\text{tok}}(c) \bigr)$
\State \Return $h'$ \Comment{lexicographic: coverage first, acoustic tie-break}
\end{algorithmic}
\end{algorithm}
